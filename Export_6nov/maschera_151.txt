Private Sub btnApriOrdini_Click()
On Error GoTo Err_btnApriOrdini_Click

    Dim stDocName As String
    Dim stLinkCriteria As String

    stDocName = "Ordini"
    
    stLinkCriteria = "[IdSedeFornitore]=" & Me![IdSedeFornitore]
    DoCmd.OpenForm stDocName, , , stLinkCriteria

    Exit Sub

Err_btnApriOrdini_Click:
    MsgBox Err.Description
    Resume Exit_btnApriOrdini_Click
    
End Sub

Private Sub Form_Current()
If IsNull(IdSedeFornitore) Then
btnApriOrdini.Visible = False
Else: btnApriOrdini.Visible = True
End If
End Sub
