Sub CasellaCombinata24_AfterUpdate()
    ' Trova il record corrispondente al controllo
    Me.RecordsetClone.FindFirst "[IdFornitore] = " & Me![CasellaCombinata24]

End Sub

Private Sub Form_Current()
If IsNull(IdFornitore) Then
btnApriSede.Visible = False
Else: btnApriSede.Visible = True
End If
CasellaCombinata24 = IdFornitore
End Sub
Private Sub btnApriSede_Click()
On Error GoTo Err_btnApriSede_Click

    Dim stDocName As String
    Dim stLinkCriteria As String

    stDocName = "Sede"
    
    stLinkCriteria = "[IdFornitore]=" & Me![IdFornitore]
    DoCmd.OpenForm stDocName, , , stLinkCriteria

Exit_btnApriSede_Click:
    Exit Sub

Err_btnApriSede_Click:
    MsgBox Err.Description
    Resume Exit_btnApriSede_Click
    
End Sub
