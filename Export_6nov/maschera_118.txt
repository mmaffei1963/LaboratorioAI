DoCmd.SetWarnings False
    DoCmd.OpenQuery "Query Correzione1", acNormal, acEdit
    DoCmd.OpenQuery "Query Correzione2", acNormal, acEdit
End Sub

Private Sub Fornitore_AfterUpdate()
[Indirizzo] = DLookup("[Anag_fornitori Query]![Indirizzo]", "[Anag_fornitori Query]", "[Anag_fornitori Query]![RAGSOC]=Forms![Ordine]![Fornitore]")
[Telefono] = DLookup("[Anag_fornitori Query]![Tel]", "[Anag_fornitori Query]", "[Anag_fornitori Query]![RAGSOC]=Forms![Ordine]![Fornitore]")
[Fax] = DLookup("[Anag_fornitori Query]![Fax]", "[Anag_fornitori Query]", "[Anag_fornitori Query]![RAGSOC]=Forms![Ordine]![Fornitore]")

End Sub


Private Sub Prezzo_Exit(Cancel As Integer)
If [Prezzo] = 0 Then
[Prezzo] = DLast("[Elenco acquisti]![Prezzo]", "[Elenco acquisti]", "[Elenco acquisti]![articolo]=Forms![Ordine]![articolo]")
End If
End Sub

Private Sub Valore_Exit(Cancel As Integer)
[Valore] = [Quantita] * [Prezzo]
End Sub
Private Sub MODIFICA_ORDINE_Click()
On Error GoTo Err_MODIFICA_ORDINE_Click

    Dim stDocName As String

    stDocName = "Query OrdineM"
    DoCmd.OpenQuery stDocName, acNormal, acEdit

Exit_MODIFICA_ORDINE_Click:
    Exit Sub

Err_MODIFICA_ORDINE_Click:
    MsgBox Err.Description
    Resume Exit_MODIFICA_ORDINE_Click
    
End Sub
